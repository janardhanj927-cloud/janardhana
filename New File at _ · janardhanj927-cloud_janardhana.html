void systemStatus()
{
bool alert = false;
bool warning = false;

// Limits from the notebook applied here
if (gasValue > 2000 || temperature > 36.0 || temperature < 18.0 || humidity > 80.0 || humidity < 30.0)
{
alert = true;
}
if (distance < 30 || rainValue < 1800 || soilValue > 3000)
{
warning = true;
}

lcd.clear();

if (alert)
{
lcd.setCursor(0, 0);
lcd.print("SYSTEM ALERT");
lcd.setCursor(0, 1);
lcd.print("CHECK NOW");
tone(BUZZER, 2200);
delay(300);
noTone(BUZZER);
delay(100);
tone(BUZZER, 2200);
delay(300);
noTone(BUZZER);
}
else if (warning)
{
lcd.setCursor(0, 0);
lcd.print("SYSTEM");
lcd.setCursor(0, 1);
lcd.print("WARNING");
tone(BUZZER, 1500);
delay(150);
noTone(BUZZER);
}
else
{
lcd.setCursor(0, 0);
lcd.print("SYSTEM SAFE");
lcd.setCursor(0, 1);
lcd.print("ALL OK");
noTone(BUZZER);
}
delay(2500);
}
